接入示例

import openai

client = openai.OpenAI(
    api_key="your-api-key",
    base_url="https://api.example.com/v1"
)

# 调用GPT-4o
resp = client.chat.completions.create(
    model="gpt-4o",
    messages=[{"role": "user", "content": "你好"}]
)

# 调用Claude - 只改model
resp = client.chat.completions.create(
    model="claude-3-5-sonnet-20241022",
    messages=[{"role": "user", "content": "你好"}]
)
